// custom orbit control
let camTarget = new THREE.Vector3(0, 1.05, 0);
let camDist = 4.2, camTheta = 0.5, camPhi = 1.25;
function updateCamera(){
  camPhi = Math.max(0.35, Math.min(1.5, camPhi));
  camDist = Math.max(2.0, Math.min(9, camDist));
  const x = camTarget.x + camDist * Math.sin(camPhi) * Math.sin(camTheta);
  const y = camTarget.y + camDist * Math.cos(camPhi);
  const z = camTarget.z + camDist * Math.sin(camPhi) * Math.cos(camTheta);
  camera.position.set(x,y,z);
  camera.lookAt(camTarget);
}
updateCamera();
let dragging = false, lastX=0, lastY=0;
renderer.domElement.addEventListener('pointerdown', e=>{ dragging=true; lastX=e.clientX; lastY=e.clientY; });
window.addEventListener('pointerup', ()=> dragging=false);
window.addEventListener('pointermove', e=>{
  if(!dragging) return;
  const dx = e.clientX-lastX, dy = e.clientY-lastY;
  lastX=e.clientX; lastY=e.clientY;
  camTheta -= dx*0.008;
  camPhi -= dy*0.008;
  updateCamera();
});
renderer.domElement.addEventListener('wheel', e=>{
  camDist += e.deltaY * 0.0025 * camDist;
  updateCamera();
  e.preventDefault();
}, { passive:false });
 
// ---------- MATERIAL HELPER ----------
function mat(color, opts={}){
  return new THREE.MeshStandardMaterial({
    color: hexToInt(color), flatShading:true,
    roughness: opts.rough !== undefined ? opts.rough : (opts.metal ? 0.35 : 0.8),
    metalness: opts.metal ? 0.65 : 0.04,
    emissive: opts.glow ? hexToInt(opts.glow) : 0x000000,
    emissiveIntensity: opts.glow ? 0.6 : 0,
  });
}
function box(w,h,d,m){ const me=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),m); me.castShadow=true; return me; }
function cyl(r1,r2,h,m,seg=6){ const me=new THREE.Mesh(new THREE.CylinderGeometry(r1,r2,h,seg),m); me.castShadow=true; return me; }
function cone(r,h,m,seg=6){ const me=new THREE.Mesh(new THREE.ConeGeometry(r,h,seg),m); me.castShadow=true; return me; }
function sph(r,m,seg=7){ const me=new THREE.Mesh(new THREE.SphereGeometry(r,seg,seg),m); me.castShadow=true; return me; }
function ico(r,m,det=0){ const me=new THREE.Mesh(new THREE.IcosahedronGeometry(r,det),m); me.castShadow=true; return me; }
function torus(r,tube,m,seg=16,arc){ const me=new THREE.Mesh(new THREE.TorusGeometry(r,tube,6,seg,arc), m); me.castShadow=true; return me; }
 
// ---------- WEAPON BUILDER ----------
// All weapons are built so the GRIP sits at local origin (0,0,0) and the
// blade/head extends in -Y (down and away from the hand), matching a
// natural relaxed hold. The hand mesh sits at the same origin as the grip.
function buildWeapon(type, accentHex, metalM){
  const g = new THREE.Group();
  g.userData.type = type;
  if(type === "none") return g;
  const metal = metalM || mat("#c9c9c9",{metal:true});
  const wood = mat("#5c3f24");
 
  switch(type){
    case "sword": {
      g.add(cyl(0.017,0.02,0.15,wood).translateY(-0.07));
      const guard = box(0.11,0.02,0.02,metal); guard.position.y=-0.145; g.add(guard);
      const blade = box(0.032,0.46,0.011,metal); blade.position.y=-0.375; g.add(blade);
      const tip = cone(0.016,0.055,metal,4); tip.rotation.x=Math.PI; tip.position.y=-0.635; g.add(tip);
      break;
    }
    case "greatsword": {
      g.add(cyl(0.024,0.026,0.26,wood).translateY(-0.12));
      const guard = box(0.2,0.03,0.03,metal); guard.position.y=-0.26; g.add(guard);
      const blade = box(0.065,0.78,0.02,metal); blade.position.y=-0.66; g.add(blade);
      const tip = cone(0.032,0.11,metal,4); tip.rotation.x=Math.PI; tip.position.y=-1.11; g.add(tip);
      break;
    }
    case "axe": {
      g.add(cyl(0.02,0.022,0.52,wood).translateY(-0.25));
      const head = box(0.05,0.26,0.2,metal); head.rotation.z=0.18; head.position.set(0.02,-0.44,0); g.add(head);
      break;
    }
    case "dagger": {
      g.add(cyl(0.014,0.016,0.09,wood).translateY(-0.045));
      const blade = box(0.024,0.18,0.01,metal); blade.position.y=-0.18; g.add(blade);
      const tip = cone(0.012,0.035,metal,4); tip.rotation.x=Math.PI; tip.position.y=-0.29; g.add(tip);
      break;
    }
    case "staff": {
      g.add(cyl(0.022,0.026,1.05,wood).translateY(-0.5));
      const gem = ico(0.075, mat(accentHex, {glow:accentHex, metal:false}), 0);
      gem.position.y = 0.06; gem.userData.isGem = true;
      g.add(gem);
      break;
    }
    case "mace": {
      g.add(cyl(0.023,0.023,0.55,wood).translateY(-0.26));
      const head = ico(0.09, metal, 0); head.position.y=-0.57; g.add(head);
      // flanges
      for(let i=0;i<4;i++){
        const f = box(0.02,0.1,0.06, metal); f.position.set(0,-0.57,0);
        f.rotation.y = i*Math.PI/2; f.position.x = Math.cos(i*Math.PI/2)*0.08; f.position.z = Math.sin(i*Math.PI/2)*0.08;
        g.add(f);
      }
      break;
    }
    case "bow": {
      const arcGeo = new THREE.TorusGeometry(0.34, 0.016, 6, 16, Math.PI*1.25);
      const bow = new THREE.Mesh(arcGeo, wood);
      bow.rotation.z = Math.PI/2 - (Math.PI*1.25)/2;
      bow.castShadow = true;
      g.add(bow);
      break;
    }
  }
  return g;
}
function buildShield(accentHex, metalM){
  const g = new THREE.Group();
  const face = cyl(0.15,0.15,0.028,metalM,8); face.rotation.z=Math.PI/2; g.add(face);
  const boss = ico(0.05, mat(accentHex,{metal:true}),0); boss.position.x=0.016; g.add(boss);
  return g;
}
 
// ---------- HAIR BUILDER ----------
function buildHair(style, hairColor, rng){
  const g = new THREE.Group();
  const m = mat(hairColor);
  if(style === "bald") return g;
  if(style === "short"){
    const dome = new THREE.Mesh(new THREE.SphereGeometry(0.19,8,6,0,Math.PI*2,0,Math.PI/1.7), m);
    dome.position.y=0.02; dome.castShadow=true; g.add(dome);
  } else if(style === "messy"){
    for(let i=0;i<7;i++){
      const t = cone(0.05+rng()*0.02, 0.1+rng()*0.05, m, 4);
      const a = (i/7)*Math.PI*2;
      t.position.set(Math.cos(a)*0.13, 0.12+rng()*0.04, Math.sin(a)*0.13);
      t.rotation.set(rng()*0.6-0.3, a, rng()*0.6-0.3);
      g.add(t);
    }
  } else if(style === "mohawk"){
    const ridge = box(0.05,0.18,0.32,m); ridge.position.y=0.16; g.add(ridge);
  } else if(style === "ponytail"){
    const dome = new THREE.Mesh(new THREE.SphereGeometry(0.19,8,6,0,Math.PI*2,0,Math.PI/1.9), m);
    dome.position.y=0.02; g.add(dome);
    const tail = cyl(0.035,0.02,0.35,m); tail.position.set(0,-0.05,-0.19); tail.rotation.x=0.5; g.add(tail);
  } else if(style === "long"){
    const dome = new THREE.Mesh(new THREE.SphereGeometry(0.19,8,6,0,Math.PI*2,0,Math.PI/1.9), m);
    dome.position.y=0.02; g.add(dome);
    const back = box(0.24,0.42,0.06,m); back.position.set(0,-0.18,-0.15); back.rotation.x=0.15; g.add(back);
  }
  return g;
}
 
// ---------- CHARACTER BUILDER ----------
let charGroup = null;
let refs = {};
 
function buildCharacter(cfg){
  if(charGroup){ scene.remove(charGroup); }
  refs = {};
  const rng = makeRNG(cfg.seed);
  const race = RACES[cfg.race];
  const clsData = classSet()[cfg.cls];
  const tierData = TIERS[cfg.tier];
  const ad = ARMOR_DEFS[cfg.armor];
 
  const skinM = mat(cfg.colors.skin, race.glow ? {glow:race.glow} : {});
  const hairM = mat(cfg.colors.hair);
  const armorM = mat(cfg.colors.armor, { metal: ad.metal, rough: ad.rough });
  const accentM = mat(cfg.colors.accent);
  const trimM = mat(ad.trim || cfg.colors.accent, {metal:true});
  const metalM = mat("#c7c2b6", {metal:true});
  const bulk = cfg.body.bulk;
 
  const root = new THREE.Group();
 
  // ----- Hips (all width/bulk propagates from here) -----
  const hips = box(0.34, 0.22, 0.24, armorM);
  hips.position.y = 0.95;
  hips.rotation.x = race.hunch*0.4;
  hips.scale.set(bulk, 1, bulk);
  root.add(hips);
 
  // ----- Spine / Chest -----
  const spine = new THREE.Group();
  spine.position.y = 0.11;
  spine.rotation.x = race.hunch;
  hips.add(spine);
  refs.spine = spine;
 
  const chest = box(0.4, 0.42, 0.26, armorM);
  chest.position.y = 0.24;
  spine.add(chest);
 
  if(ad.plate){
    const plate = box(0.42, 0.22, 0.02, metalM);
    plate.position.set(0, 0.32, 0.14);
    spine.add(plate);
  }
  if(ad.scales){
    for(let row=0; row<2; row++){
      for(let i=0;i<3;i++){
        const s = ico(0.055, armorM, 0);
        s.scale.set(1,1,0.35);
        s.position.set((i-1)*0.13, 0.2+row*0.13, 0.15);
        spine.add(s);
      }
    }
  }
  if(ad.pauldron){
    const lp = ico(0.1, metalM, 0); lp.position.set(0.25, 0.42, 0); spine.add(lp);
    const rp = ico(0.1, metalM, 0); rp.position.set(-0.25, 0.42, 0); spine.add(rp);
  }
  if(ad.cape){
    const cape = box(0.34, 0.55, 0.02, cfg.armor==='royal' ? accentM : armorM);
    cape.position.set(0, 0.02, -0.16);
    cape.rotation.x = -0.18;
    spine.add(cape);
  }
  if(ad.helm){
    const helm = new THREE.Mesh(new THREE.SphereGeometry(0.16,8,6,0,Math.PI*2,0,Math.PI/1.8), metalM);
    helm.position.set(0,0.62,0);
    helm.castShadow = true;
    spine.add(helm);
    const visor = box(0.2,0.05,0.04, mat("#1a1410"));
    visor.position.set(0,0.6,0.13); spine.add(visor);
  }
  if(ad.trinkets){
    const feather = cone(0.025,0.16, mat("#c9a13a"), 4);
    feather.position.set(0.24,0.48,-0.02); feather.rotation.z=-0.5;
    spine.add(feather);
  }
  if(ad.trim && ad.cape){
    const hem = torus(0.19,0.012,trimM,16); hem.rotation.x=Math.PI/2; hem.position.y=0.03; spine.add(hem);
  }
 
  // ----- Robe (mage / monk classes replace lower torso + legs read) -----
  if(ad.robe){
    const style = ad.robe;
    let topR, botR, h, hemColor;
    if(style === "mage"){ topR=0.24; botR=0.52; h=1.05; }
    else { topR=0.23; botR=0.36; h=0.8; } // monk
    const robe = cyl(topR*bulk, botR*bulk, h, mat(cfg.colors.armor));
    robe.position.y = -h/2 - 0.08;
    spine.add(robe);
 
    // belt / trim
    const belt = torus(topR*bulk*0.72, 0.016, style==='mage' ? accentM : mat("#4a3a2a"), 16);
    belt.rotation.x = Math.PI/2; belt.position.y = 0.13;
    spine.add(belt);
 
    if(style === "mage"){
      const hem = torus(botR*bulk*0.98, 0.014, accentM, 16);
      hem.rotation.x = Math.PI/2; hem.position.y = -h-0.06;
      spine.add(hem);
      // pointed wizard hat
      if(cfg.cls === "mage" || cfg.cls === "shaman"){
        const hatGroup = new THREE.Group();
        hatGroup.position.y = 0.78;
        const brim = cyl(0.22,0.22,0.02, mat(cfg.colors.armor)); brim.position.y=0; hatGroup.add(brim);
        const cap = cone(0.16,0.42, mat(cfg.colors.armor), 6); cap.position.y=0.24; cap.rotation.z=0.12; hatGroup.add(cap);
        const band = torus(0.155,0.013,accentM,12); band.rotation.x=Math.PI/2; band.position.y=0.04; hatGroup.add(band);
        spine.add(hatGroup);
        refs.hat = hatGroup;
      }
    } else {
      const rope = torus(topR*bulk*0.74, 0.012, mat("#c9a877"), 16);
      rope.rotation.x = Math.PI/2; rope.position.y = 0.1;
      spine.add(rope);
    }
  }
 
  // ----- Neck / Head -----
  const neck = cyl(0.055,0.06,0.08, skinM);
  neck.position.y = 0.46;
  spine.add(neck);
 
  const headGroup = new THREE.Group();
  headGroup.position.y = 0.62;
  const headScale = cfg.body.head * race.headM;
  headGroup.scale.set(headScale/bulk, headScale, headScale/bulk); // cancel inherited torso bulk so head size stays independent
  spine.add(headGroup);
  refs.head = headGroup;
 
  const headShape = cfg.race === "goblin" ? new THREE.IcosahedronGeometry(0.155,0) : new THREE.BoxGeometry(0.24,0.24,0.22);
  const head = new THREE.Mesh(headShape, skinM);
  head.castShadow = true;
  headGroup.add(head);
 
  const eyeColor = race.glow ? mat(race.glow,{glow:race.glow}) : mat("#1a1410");
  const le = sph(0.018, eyeColor, 5); le.position.set(0.07,0.02,0.115); headGroup.add(le);
  const re = sph(0.018, eyeColor, 5); re.position.set(-0.07,0.02,0.115); headGroup.add(re);
 
  if(race.ears){
    const le2 = cone(0.035,0.13, skinM, 4); le2.position.set(0.15,0.06,0); le2.rotation.z=-1.1; headGroup.add(le2);
    const re2 = cone(0.035,0.13, skinM, 4); re2.position.set(-0.15,0.06,0); re2.rotation.z=1.1; headGroup.add(re2);
  }
  if(race.tusks){
    const lt = cone(0.018,0.09, mat("#e8e2d0"), 4); lt.position.set(0.06,-0.11,0.08); lt.rotation.x=2.6; headGroup.add(lt);
    const rt = cone(0.018,0.09, mat("#e8e2d0"), 4); rt.position.set(-0.06,-0.11,0.08); rt.rotation.x=2.6; headGroup.add(rt);
  }
  if(race.horns){
    const lh = cone(0.03,0.18, mat("#2a1a14"), 4); lh.position.set(0.08,0.15,-0.02); lh.rotation.set(-0.4,0,-0.3); headGroup.add(lh);
    const rh = cone(0.03,0.18, mat("#2a1a14"), 4); rh.position.set(-0.08,0.15,-0.02); rh.rotation.set(-0.4,0,0.3); headGroup.add(rh);
  }
  if(race.beard || cfg.cls === "cleric"){
    const beard = cone(0.09,0.16, hairM, 5); beard.position.set(0,-0.16,0.04); headGroup.add(beard);
  }
  if(!ad.robe || cfg.cls !== "mage"){
    const hair = buildHair(cfg.hair, cfg.colors.hair, rng);
    hair.position.y = 0.02;
    headGroup.add(hair);
  }
 
  // ----- Arms -----
  function buildArm(side){
    const sign = side === "L" ? 1 : -1;
    const shoulder = new THREE.Group();
    shoulder.position.set(sign*0.24*cfg.body.shoulders, 0.4, 0);
    shoulder.scale.y = cfg.body.arms;
    spine.add(shoulder);
 
    const upper = cyl(0.05,0.045,0.26, skinM);
    upper.position.y = -0.13;
    upper.rotation.z = sign*0.06;
    shoulder.add(upper);
 
    const forearmGroup = new THREE.Group();
    forearmGroup.position.y = -0.26;
    upper.add(forearmGroup);
    const forearm = cyl(0.042,0.036,0.24, skinM);
    forearm.position.y = -0.12;
    forearmGroup.add(forearm);
 
    const handGroup = new THREE.Group();
    handGroup.position.y = -0.24;
    forearmGroup.add(handGroup);
    const hand = sph(0.05, skinM, 5);
    handGroup.add(hand);
 
    return { shoulder, forearmGroup, handGroup };
  }
  const armL = buildArm("L");
  const armR = buildArm("R");
  refs.leftUpperArm = armL.shoulder;
  refs.rightUpperArm = armR.shoulder;
  refs.leftHand = armL.handGroup;
  refs.rightHand = armR.handGroup;
 
  // weapon: bows sit naturally in the off-hand; everything else in the main hand.
  const weaponG = buildWeapon(cfg.weapon, cfg.colors.accent, metalM);
  if(cfg.weapon === "bow"){
    weaponG.rotation.z = Math.PI/2;
    armL.handGroup.add(weaponG);
  } else if(cfg.weapon !== "none"){
    armR.handGroup.add(weaponG);
  }
  refs.gem = weaponG.getObjectByProperty ? null : null;
  weaponG.traverse(o=>{ if(o.userData && o.userData.isGem) refs.gem = o; });
 
  if(clsData.offhand === "shield"){
    const shield = buildShield(cfg.colors.accent, metalM);
    armL.handGroup.add(shield);
  } else if(clsData.offhand === "dagger"){
    const off = buildWeapon("dagger", cfg.colors.accent, metalM);
    armL.handGroup.add(off);
  }
 
  // ----- Legs -----
  function buildLeg(side){
    const sign = side === "L" ? 1 : -1;
    const thighGroup = new THREE.Group();
    thighGroup.position.set(sign*0.1, -0.11, 0);
    thighGroup.scale.set(bulk, cfg.body.legs, bulk);
    hips.add(thighGroup);
    const thigh = cyl(0.06,0.055,0.32, mat(cfg.colors.armor));
    thigh.position.y = -0.16;
    thighGroup.add(thigh);
 
    const shinGroup = new THREE.Group();
    shinGroup.position.y = -0.32;
    thigh.add(shinGroup);
    const shin = cyl(0.05,0.04,0.3, skinM);
    shin.position.y = -0.15;
    shinGroup.add(shin);
 
    const foot = box(0.09,0.06,0.16, mat("#2a2018"));
    foot.position.set(0,-0.33,0.03);
    shinGroup.add(foot);
 
    return { thighGroup };
  }
  const legL = buildLeg("L");
  const legR = buildLeg("R");
  refs.leftThigh = legL.thighGroup;
  refs.rightThigh = legR.thighGroup;
 
  if(race.tail){
    const tail = new THREE.Group();
    tail.position.set(0,-0.05,-0.14);
    hips.add(tail);
    let prev = tail;
    for(let i=0;i<3;i++){
      const seg = cyl(0.03-i*0.006,0.026-i*0.006,0.18, skinM);
      seg.position.y = -0.09;
      seg.rotation.x = -0.35;
      prev.add(seg);
      const next = new THREE.Group(); next.position.y=-0.18; seg.add(next);
      prev = next;
    }
  }
 
  // overall height (vertical stretch only) + tier size
  root.scale.set(tierData.scale, tierData.scale*cfg.body.height, tierData.scale);
 
  const glowAmt = tierData.glow;
  runeMat.emissiveIntensity = 0.4 + glowAmt;
  runeMat2.emissiveIntensity = 0.25 + glowAmt*0.7;
 
  root.traverse(o=>{ if(o.isMesh){ o.castShadow=true; } });
  charGroup = root;
  scene.add(root);
}
 
// ---------- ANIMATION ----------
let animState = "idle";
let animClock = 0;
 
function animate(dt){
  animClock += dt;
  if(!refs.spine) return;
  const t = animClock;
 
  charGroup.position.set(0,0,0);
  charGroup.rotation.set(0,0,0);
  refs.spine.rotation.y = 0;
  if(refs.gem) refs.gem.material.emissiveIntensity = 0.6;
 
  if(animState === "idle"){
    const sway = Math.sin(t*1.6);
    refs.spine.rotation.z = sway*0.02;
    refs.spine.rotation.x = (RACES[state.race].hunch) + sway*0.008;
    refs.head.rotation.y = Math.sin(t*0.5)*0.12;
    refs.leftUpperArm.rotation.x = Math.sin(t*1.6)*0.05;
    refs.rightUpperArm.rotation.x = -Math.sin(t*1.6)*0.05 + 0.05;
    refs.leftThigh.rotation.x = 0;
    refs.rightThigh.rotation.x = 0;
  } else if(animState === "walk" || animState === "run"){
    const speed = animState === "walk" ? 5 : 9;
    const amp = animState === "walk" ? 0.5 : 0.85;
    refs.leftUpperArm.rotation.x = Math.sin(t*speed)*amp;
    refs.rightUpperArm.rotation.x = -Math.sin(t*speed)*amp;
    refs.leftThigh.rotation.x = -Math.sin(t*speed)*amp;
    refs.rightThigh.rotation.x = Math.sin(t*speed)*amp;
    refs.spine.rotation.z = Math.sin(t*speed)*0.06;
    refs.head.rotation.y = Math.sin(t*speed*0.5)*0.08;
    charGroup.position.y = Math.abs(Math.sin(t*speed))*0.04;
  } else if(animState === "attack"){
    const cycle = (t % 0.7)/0.7;
    const swing = Math.sin(cycle*Math.PI);
    refs.rightUpperArm.rotation.x = -1.5*swing;
    refs.rightUpperArm.rotation.z = -0.3*swing;
    refs.spine.rotation.y = 0.15*swing;
    refs.leftThigh.rotation.x = 0.15*swing;
    refs.rightThigh.rotation.x = -0.1*swing;
  } else if(animState === "block"){
    const raise = Math.min(1, t*4);
    refs.leftUpperArm.rotation.x = -1.1*raise;
    refs.leftUpperArm.rotation.z = 0.2*raise;
    refs.rightUpperArm.rotation.x = -0.9*raise;
    refs.spine.rotation.x = (RACES[state.race].hunch) - 0.08*raise;
    refs.leftThigh.rotation.x = 0.15*raise;
    refs.rightThigh.rotation.x = 0.15*raise;
  } else if(animState === "cast"){
    const pulse = (Math.sin(t*3)+1)/2;
    refs.leftUpperArm.rotation.x = -1.3 - pulse*0.1;
    refs.rightUpperArm.rotation.x = -1.3 - pulse*0.1;
    refs.rightUpperArm.rotation.z = -0.35;
    refs.leftUpperArm.rotation.z = 0.35;
    refs.spine.rotation.x = (RACES[state.race].hunch) - 0.1;
    refs.head.rotation.x = -0.1;
    if(refs.gem) refs.gem.material.emissiveIntensity = 0.9 + pulse*1.2;
  } else if(animState === "hit"){
    const cycle = (t % 0.5)/0.5;
    const snap = Math.sin(cycle*Math.PI);
    refs.spine.rotation.x = (RACES[state.race].hunch) - 0.35*snap;
    refs.head.rotation.x = -0.3*snap;
    refs.leftUpperArm.rotation.x = 0.3*snap;
    refs.rightUpperArm.rotation.x = 0.3*snap;
    charGroup.position.z = 0.06*snap;
  } else if(animState === "cheer"){
    const bounce = Math.abs(Math.sin(t*5));
    refs.leftUpperArm.rotation.x = -2.4 + Math.sin(t*5)*0.15;
    refs.rightUpperArm.rotation.x = -2.4 - Math.sin(t*5)*0.15;
    refs.leftUpperArm.rotation.z = 0.3;
    refs.rightUpperArm.rotation.z = -0.3;
    charGroup.position.y = bounce*0.12;
    refs.head.rotation.y = Math.sin(t*3)*0.15;
  } else if(animState === "sit"){
    refs.leftThigh.rotation.x = -1.4;
    refs.rightThigh.rotation.x = -1.4;
    refs.spine.rotation.x = (RACES[state.race].hunch) + 0.15;
    charGroup.position.y = -0.42*state.body.height;
    refs.leftUpperArm.rotation.x = Math.sin(t*1.2)*0.04;
    refs.rightUpperArm.rotation.x = Math.sin(t*1.2)*0.04 + 0.1;
  } else if(animState === "sleep"){
    charGroup.rotation.z = Math.PI/2;
    charGroup.position.y = -0.05 + Math.sin(t*1.2)*0.015;
    charGroup.position.x = -0.35;
    refs.leftUpperArm.rotation.x = 0.1;
    refs.rightUpperArm.rotation.x = 0.1;
  } else if(animState === "death"){
    const p = Math.min(1, t*1.2);
    charGroup.rotation.x = p * Math.PI/2.1;
    charGroup.position.y = -p*0.15;
  }
}
 
// ---------- RENDER LOOP ----------
let lastTime = performance.now();
function loop(){
  requestAnimationFrame(loop);
  const now = performance.now();
  const dt = Math.min(0.05,(now-lastTime)/1000);
  lastTime = now;
  animate(dt);
  renderer.render(scene, camera);
}
loop();
 
window.addEventListener('resize', ()=>{
  camera.aspect = viewport.clientWidth/viewport.clientHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(viewport.clientWidth, viewport.clientHeight);
});

function buildOptionGrid(el, items, group, labelFn){
  el.innerHTML = "";
  items.forEach(key=>{
    const btn = document.createElement('button');
    btn.className = 'opt';
    btn.textContent = labelFn ? labelFn(key) : niceLabel(key);
    btn.dataset.group = group;
    btn.dataset.value = key;
    btn.addEventListener('click', ()=>{
      el.querySelectorAll('button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      onControlChange(group, key);
    });
    el.appendChild(btn);
  });
}
function setActiveInGrid(el, value){
  el.querySelectorAll('button').forEach(b=> b.classList.toggle('active', b.dataset.value === value));
}

function refreshClassSelect(){
  const sel = document.getElementById('classSelect');
  const src = classSet();
  sel.innerHTML = "";
  const groups = {};
  Object.keys(src).forEach(k=>{
    const g = src[k].group || "Other";
    (groups[g] = groups[g]||[]).push(k);
  });
  Object.keys(groups).forEach(g=>{
    const og = document.createElement('optgroup'); og.label = g;
    groups[g].forEach(k=>{
      const o = document.createElement('option'); o.value=k; o.textContent = src[k].label;
      og.appendChild(o);
    });
    sel.appendChild(og);
  });
  if(!src[state.cls]) state.cls = Object.keys(src)[0];
  sel.value = state.cls;
}

function applyRaceDefaults(){

function onControlChange(group, value){
  if(group === 'race'){ state.race = value; applyRaceDefaults(); }
  else if(group === 'tier'){ state.tier = value; }

  else if(group === 'armor'){ state.armor = value; document.getElementById('c_armor').value = ARMOR_DEFS[value].tone; state.colors.armor = ARMOR_DEFS[value].tone; }
  else if(group === 'weapon'){ state.weapon = value; }
  rebuild();
}
